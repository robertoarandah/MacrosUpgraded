Limpieza de formato básica let cleanData = data.trim().replace(/}\s*{/g, '},{').replace(/[“”]/g, '"'); if (!cleanData.startsWith('[')) cleanData = '[' + cleanData; if (!cleanData.endsWith(']')) cleanData = cleanData + ']'; // 2. Intentar leer la lista completa let temp = JSON.parse(cleanData); // 3. Aplanar en caso de que lleguen arreglos dobles [[{...}]] parsed = Array.isArray(temp) ? temp.flat(Infinity) : [temp]; } catch (e) { // MODO DE EMERGENCIA: Si el atajo cortó el texto y el JSON está roto, // extraemos cada alimento individualmente usando Expresiones Regulares. console.warn("JSON roto, usando extracción de emergencia."); let matches = data.match(/\{[^{}]+\}/g); if (matches) { matches.forEach(m => { try { parsed.push(JSON.parse(m.replace(/[“”]/g, '"'))); } catch(err) {} }); } } // 4. Guardar los alimentos rescatados if (parsed && parsed.length > 0) { let targetMeal = 'desayuno'; const hour = new Date().getHours(); if (hour >= 12 && hour < 18) targetMeal = 'almuerzo'; else if (hour >= 18) targetMeal = 'cena'; let count = 0; parsed.forEach(item => { // Validar que el alimento tenga nombre para evitar celdas vacías if (item && item.name) { const newEntry = { id: Date.now() + Math.floor(Math.random() * 1000) + count, meal: targetMeal, name: item.name, cal: Math.round(item.cal || 0), pro: Math.round(item.pro || 0), car: Math.round(item.car || 0), fat: Math.round(item.fat || 0) }; entries.push(newEntry); count++; } }); if (count > 0) { saveData(); showToast('✓ ' + count + ' alimentos añadidos'); } else { showToast('No se encontraron alimentos legibles.'); } } // 5.
